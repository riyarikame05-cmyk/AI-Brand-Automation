def get_local_path(
    asset: Asset,
):

    if not asset.url:

        return None


    if "/generated/" not in asset.url:

        return None


    filename = (
        asset.url
        .split("/generated/")[-1]
    )


    return (
        GENERATED_DIR
        / filename
    )


# =========================================================
# COLOR HELPER
# =========================================================

def extract_palette(
    file_path: Path,
):

    colors = (
        extract_dominant_colors(
            file_path,
            number_of_colors=3,
        )
    )


    if not colors:

        raise ValueError(
            "No colors were extracted."
        )


    # Make extraction resilient in case the
    # image contains fewer than three colors.
    colors = list(colors)


    while len(colors) < 3:

        colors.append(
            colors[-1]
        )


    return {
        "primary_color":
            colors[0],

        "secondary_color":
            colors[1],

        "accent_color":
            colors[2],
    }

def create_logo_concepts(
    project: Project,
    db: Session,
    count: int = 3,
):

    concepts = []

    created_files = []


    try:

        for number in range(count):

            print(
                f"[LOGO] Generating "
                f"{number + 1}/{count} "
                f"for {project.business_name}"
            )


            # =============================================
            # GENERATE IMAGE
            # =============================================

            image_bytes = (
                generate_logo_image(
                    business_name=(
                        project.business_name
                    ),

                    industry=(
                        project.industry
                    ),

                    style=(
                        project.style
                    ),

                    description=(
                        project.description
                    ),
                )
            )


            if not isinstance(
                image_bytes,
                (bytes, bytearray),
            ):

                raise RuntimeError(
                    "Logo generator returned "
                    "invalid image data."
                )


            if len(image_bytes) < 1000:

                raise RuntimeError(
                    "Generated logo image "
                    "is too small or invalid."
                )


            # =============================================
            # SAVE IMAGE
            # =============================================

            filename = (
                f"{project.id}-"
                f"{uuid.uuid4()}.png"
            )


            file_path = (
                GENERATED_DIR
                / filename
            )


            file_path.write_bytes(
                bytes(image_bytes)
            )


            created_files.append(
                file_path
            )


            image_url = (
                "http://127.0.0.1:8000"
                f"/generated/{filename}"
            )


            # =============================================
            # EXTRACT COLORS
            # =============================================

            try:

                palette = (
                    extract_palette(
                        file_path
                    )
                )


            except Exception as error:

                print(
                    "[COLOR] Extraction failed:",
                    repr(error),
                )


                # Palette extraction should not destroy
                # an otherwise valid generated logo.
                palette = {
                    "primary_color":
                        "#111827",

                    "secondary_color":
                        "#6366F1",

                    "accent_color":
                        "#8B5CF6",
                }


            # =============================================
            # SAVE DATABASE ASSET
            # =============================================

            asset = Asset(
                project_id=(
                    project.id
                ),

                asset_type="logo",

                url=(
                    image_url
                ),

                prompt=(
                    f"Logo concept "
                    f"{number + 1} for "
                    f"{project.business_name}"
                ),

                is_selected=False,
            )


            db.add(
                asset
            )

            db.flush()


            concepts.append({
                "asset":
                    asset,

                "palette":
                    palette,
            })


            print(
                f"[LOGO] Completed "
                f"{number + 1}/{count}"
            )


        return concepts


    except Exception:

        # SQLAlchemy rollback happens in the
        # route calling this helper.
        #
        # Delete generated files so failed
        # generations do not leave orphan files.

        for file_path in created_files:

            try:

                if file_path.exists():

                    file_path.unlink()

            except OSError:

                pass


        raise

    project_id: str,

    db: Session = Depends(get_db),
):

    # =====================================================
    # PROJECT
    # =====================================================

    project = (
        db.query(Project)
        .filter(
            Project.id
            == project_id
        )
        .first()
    )


    if not project:

        raise HTTPException(
            status_code=404,
            detail=(
                "Project not found."
            ),
        )


    # =====================================================
    # LOGO
    # =====================================================

    logo = (
        db.query(Asset)
        .filter(
            Asset.id
            == asset_id,

            Asset.project_id
            == project.id,

            Asset.asset_type
            == "logo",
        )
        .first()
    )


    if not logo:

        raise HTTPException(
            status_code=404,
            detail=(
                "Logo not found."
            ),
        )


    logo_path = (
        get_local_path(
            logo
        )
    )


    if (
        logo_path is None
        or not logo_path.exists()
    ):

        raise HTTPException(
            status_code=400,
            detail=(
                "Logo file not found."
            ),
        )


    try:

        # =================================================
        # SELECT LOGO
        # =================================================

        (
            db.query(Asset)
            .filter(
                Asset.project_id
                == project.id,

                Asset.asset_type
                == "logo",
            )
            .update(
                {
                    Asset.is_selected:
                        False
                },

                synchronize_session=False,
            )
        )


        logo.is_selected = True


        # =================================================
        # EXTRACT FINAL COLORS
        # =================================================

        palette = (
            extract_palette(
                logo_path
            )
        )


        # =================================================
        # LOAD BRAND KIT
        # =================================================

        brand_kit = (
            db.query(BrandKit)
            .filter(
                BrandKit.project_id
                == project.id
            )
            .first()
        )


        if not brand_kit:

            brand_kit = BrandKit(
                project_id=(
                    project.id
                ),

                primary_color=(
                    palette[
                        "primary_color"
                    ]
                ),

                secondary_color=(
                    palette[
                        "secondary_color"
                    ]
                ),

                accent_color=(
                    palette[
                        "accent_color"
                    ]
                ),

                slogan="",

                personality=(
                    json.dumps([])
                ),

                tone="",

                target_audience="",

                marketing_message="",

                heading_font=(
                    "Montserrat"
                ),

                body_font=(
                    "Inter"
                ),
            )


            db.add(
                brand_kit
            )


        else:

            brand_kit.primary_color = (
                palette[
                    "primary_color"
                ]
            )


            brand_kit.secondary_color = (
                palette[
                    "secondary_color"
                ]
            )


            brand_kit.accent_color = (
                palette[
                    "accent_color"
                ]
            )


        # =================================================
        # PERSONALITY
        # =================================================

        try:

            personality = (
                json.loads(
                    brand_kit.personality
                )
                if brand_kit.personality
                else []
            )


            if not isinstance(
                personality,
                list,
            ):

                personality = []


        except Exception:

            personality = []


        # =================================================
        # GENERATE VIDEO
        # =================================================

        print(
            "[VIDEO] Generating video..."
        )


        video_path = (
            generate_promotional_video(
                logo_path=(
                    logo_path
                ),

                output_directory=(
                    GENERATED_DIR
                ),

                business_name=(
                    project.business_name
                ),

                slogan=(
                    brand_kit.slogan
                    or ""
                ),

                personality=(
                    personality
                ),

                marketing_message=(
                    brand_kit.marketing_message
                    or ""
                ),

                primary_color=(
                    brand_kit.primary_color
                ),

                secondary_color=(
                    brand_kit.secondary_color
                ),

                accent_color=(
                    brand_kit.accent_color
                ),
            )
        )


        if not video_path:

            raise RuntimeError(
                "Video generator did not "
                "return a file."
            )


        video_path = (
            Path(video_path)
        )


        if not video_path.exists():

            raise RuntimeError(
                "Generated video file "
                "does not exist."
            )


        # =================================================
        # VIDEO ASSET
        # =================================================

        video_url = (
            "http://127.0.0.1:8000"
            f"/generated/{video_path.name}"
        )


        video = Asset(
            project_id=(
                project.id
            ),

            asset_type="video",

            url=(
                video_url
            ),

            prompt=(
                "Automatic promotional "
                "video from selected logo "
                "and brand intelligence"
            ),

            is_selected=False,
        )


        db.add(
            video
        )


        db.commit()


        db.refresh(

            "[VIDEO] Video completed."

@app.get(
    "/projects/{project_id}/assets"
)
def get_assets(
    project_id: str,
    db: Session = Depends(get_db),
):

    project = (
        db.query(Project)
        .filter(
            Project.id
            == project_id
        )
        .first()
    )


    if not project:

        raise HTTPException(
            status_code=404,
            detail=(
                "Project not found."
            ),
        )


    assets = (
        db.query(Asset)
        .filter(
            Asset.project_id
            == project.id
        )
        .order_by(
            Asset.created_at.desc()
        )
        .all()
    )


    result = []


    for asset in assets:

        data = (
            asset_to_dict(
                asset
            )
        )


        # =================================================
        # ADD PALETTE TO LOGOS
        # =================================================

        if (
            asset.asset_type
            == "logo"
        ):

            file_path = (
                get_local_path(
                    asset
                )
            )


            if (
                file_path
                and file_path.exists()
            ):

                try:

                    data["palette"] = (
                        extract_palette(
                            file_path
                        )
                    )


                except Exception as error:

                    print(
                        "[COLOR] Asset palette error:",
                        repr(error),
                    )


                    data["palette"] = None


            else:

                data["palette"] = None


        result.append(
            data
        )


    return {
        "assets":
            result
    }

@app.post(
    "/projects/{project_id}/generate-image"
)
def generate_campaign_image(
    project_id: str,
    request: CampaignImageRequest,
    db: Session = Depends(get_db),
):

    file_path = None


    # =====================================================
    # PROJECT
    # =====================================================

    project = (
        db.query(Project)
        .filter(
            Project.id
            == project_id
        )
        .first()
    )


    if not project:

        raise HTTPException(
            status_code=404,
            detail=(
                "Project not found."
            ),
        )


    try:

        # =================================================
        # BRAND KIT
        # =================================================

        brand_kit = (
            db.query(BrandKit)
            .filter(
                BrandKit.project_id
                == project.id
            )
            .first()
        )


        primary_color = (
            brand_kit.primary_color
            if brand_kit
            else "#111827"
        )


        secondary_color = (
            brand_kit.secondary_color
            if brand_kit
            else "#4F46E5"
        )


        accent_color = (
            brand_kit.accent_color
            if brand_kit
            else "#06B6D4"
        )


        # =================================================
        # PROMPT
        # =================================================

        custom_prompt = (
            request.prompt.strip()
        )


        if custom_prompt:

            concept = (
                custom_prompt
            )


        else:

            concept = (
                "premium marketing campaign "
                f"visual for "
                f"{project.business_name}, "
                f"{project.description}"
            )


        specification = {
            "concept":
                concept,

            "logo_type":
                "abstract",

            "shape":
                "dynamic",

            "primary_color":
                primary_color,

            "secondary_color":
                secondary_color,

            "accent_color":
                accent_color,
        }


        # =================================================
        # GENERATE IMAGE
        # =================================================

        image_bytes = (
            generate_image(
                provider=(
                    request.provider
                ),

                purpose=(
                    "campaign"
                ),

                specification=(
                    specification
                ),

                industry=(
                    project.industry
                ),

                style=(
                    project.style
                ),
            )
        )


        if not isinstance(
            image_bytes,
            (bytes, bytearray),
        ):

            raise RuntimeError(
                "Image provider returned "
                "invalid image data."
            )


        if len(image_bytes) < 1000:

            raise RuntimeError(
                "Generated image is invalid."
            )


        # =================================================
        # SAVE FILE
        # =================================================

        filename = (
            f"{project.id}-"
            f"campaign-"
            f"{uuid.uuid4()}.png"
        )


        file_path = (
            GENERATED_DIR
            / filename
        )


        file_path.write_bytes(
            bytes(image_bytes)
        )


        image_url = (
            "http://127.0.0.1:8000"
            f"/generated/{filename}"
        )


        # =================================================
        # DATABASE ASSET
        # =================================================

        asset = Asset(
            project_id=(
                project.id
            ),

            asset_type=(
                "campaign_image"
            ),

            url=(
                image_url
            ),

            prompt=(
                concept
            ),

            is_selected=False,
        )


        db.add(
            asset
        )


        db.commit()


        db.refresh(
            asset
        )


        return {
            "message":
                (
                    "Campaign image "
                    "generated successfully."
                ),

            "provider":
                request.provider,

            "asset":
                asset_to_dict(
                    asset
                ),
        }


    except HTTPException:

        db.rollback()

        raise


    except Exception as error:

        db.rollback()


        # =================================================
        # DELETE ORPHAN FILE
        # =================================================

        if (
            file_path is not None
            and file_path.exists()
        ):

            try:

                file_path.unlink()

            except OSError:

                pass


        print(
            "[CAMPAIGN] Generation error:",
            repr(error),
        )


        raise HTTPException(
            status_code=500,
            detail=(
                "Campaign image "
                "generation failed."
            ),
        )

@app.get(
    "/projects/{project_id}/download-package"
)
def download_brand_package(
    project_id: str,
    db: Session = Depends(get_db),
):

    # =====================================================
    # PROJECT
    # =====================================================

    project = (
        db.query(Project)
        .filter(
            Project.id
            == project_id
        )
        .first()
    )


    if not project:

        raise HTTPException(
            status_code=404,
            detail=(
                "Project not found."
            ),
        )


    # =====================================================
    # BRAND KIT
    # =====================================================

    brand_kit = (
        db.query(BrandKit)
        .filter(
            BrandKit.project_id
            == project.id
        )
        .first()
    )


    if not brand_kit:

        raise HTTPException(
            status_code=400,
            detail=(
                "Brand kit has not "
                "been generated."
            ),
        )


    # =====================================================
    # SELECTED LOGO
    # =====================================================

    selected_logo = (
        db.query(Asset)
        .filter(
            Asset.project_id
            == project.id,

            Asset.asset_type
            == "logo",

            Asset.is_selected.is_(
                True
            ),
        )
        .first()
    )


    if not selected_logo:

        raise HTTPException(
            status_code=400,
            detail=(
                "Select a logo first."
            ),
        )


    logo_path = (
        get_local_path(
            selected_logo
        )
    )


    if (
        logo_path is None
        or not logo_path.exists()
    ):

        raise HTTPException(
            status_code=400,
            detail=(
                "Selected logo file "
                "was not found."
            ),
        )


    # =====================================================
    # LATEST VIDEO
    # =====================================================

    latest_video = (
        db.query(Asset)
        .filter(
            Asset.project_id
            == project.id,

            Asset.asset_type
            == "video",
        )
        .order_by(
            Asset.created_at.desc()
        )
        .first()
    )


    video_path = None


    if latest_video:

        candidate = (
            get_local_path(
                latest_video
            )
        )


        if (
            candidate
            and candidate.exists()
        ):

            video_path = (
                candidate
            )


    # =====================================================
    # CREATE ZIP PACKAGE
    # =====================================================

    try:

        package_path = (
            create_brand_package(
                project=(
                    project
                ),

                brand_kit=(
                    brand_kit
                ),

                logo_path=(
                    logo_path
                ),

                video_path=(
                    video_path
                ),

                output_directory=(
                    GENERATED_DIR
                ),
            )
        )


        package_path = (
            Path(package_path)
        )


        if not package_path.exists():

            raise RuntimeError(
                "Brand package was not created."
            )


        return FileResponse(
            path=str(
                package_path
            ),

            filename=(
                package_path.name
            ),

            media_type=(
                "application/zip"
            ),
        )


    except Exception as error:

        print(
            "[PACKAGE] Error:",
            repr(error),
        )


        raise HTTPException(
            status_code=500,
            detail=(
                "Could not create "
                "brand package."
            ),
        )

import json
import requests


OLLAMA_URL = "http://127.0.0.1:11434/api/generate"
OLLAMA_MODEL = "qwen2.5:3b"


def generate_brand_intelligence(
    business_name: str,
    industry: str,
    description: str,
    style: str,
) -> dict:
    """
    Generate structured brand strategy using
    the locally running Ollama model.
    """

    prompt = f"""
You are a professional brand strategist.

Create a concise brand identity for this business.

Business name: {business_name}
Industry: {industry}
Business description: {description}
Brand style: {style}

Return ONLY valid JSON.

Use exactly this structure:

{{
    "slogan": "short professional slogan",
    "personality": [
        "trait 1",
        "trait 2",
        "trait 3"
    ],
    "tone": "short tone of voice description",
    "target_audience": "short target audience description",
    "marketing_message": "one short professional marketing message",
    "typography": {{
        "heading": "font recommendation",
        "body": "font recommendation"
    }}
}}

Rules:
- slogan must be 2 to 7 words
- do not include markdown
- do not include explanations
- do not invent factual claims
- personality must contain exactly 3 traits
- typography should use common professional fonts
"""

    response = requests.post(
        OLLAMA_URL,
        json={
            "model": OLLAMA_MODEL,
            "prompt": prompt,
            "stream": False,
            "format": "json",
            "options": {
                "temperature": 0.7,
            },
        },
        timeout=120,
    )

    response.raise_for_status()

    result = response.json()

    generated_text = result.get(
        "response",
        "{}",
    )

    intelligence = json.loads(
        generated_text
    )

    return {
        "slogan": intelligence.get(
            "slogan",
            ""
        ),
        "personality": intelligence.get(
            "personality",
            []
        ),
        "tone": intelligence.get(
            "tone",
            ""
        ),
        "target_audience": intelligence.get(
            "target_audience",
            ""
        ),
        "marketing_message": intelligence.get(
            "marketing_message",
            ""
        ),
        "typography": intelligence.get(
            "typography",
            {
                "heading": "Montserrat",
                "body": "Inter",
            },
        ),
    }

def polygon_points(
    cx,
    cy,
    radius,
    sides,
    rotation=0,
):

    points = []


    for index in range(sides):

        angle = math.radians(
            rotation
            + index
            * (360 / sides)
            - 90
        )


        x = (
            cx
            + radius
            * math.cos(angle)
        )


        y = (
            cy
            + radius
            * math.sin(angle)
        )


        points.append(
            (
                x,
                y,
            )
        )


    return points


def draw_center_text(
    draw,
    text,
    center_x,
    y,
    font,
    color,
):

    box = draw.textbbox(
        (
            0,
            0,
        ),
        text,
        font=font,
    )


    width = (
        box[2]
        - box[0]
    )


    draw.text(
        (
            center_x
            - width / 2,
            y,
        ),
        text,
        font=font,
        fill=color,
    )


def initials_from_name(
    business_name: str,
):

    words = [
        word
        for word
        in business_name.strip().split()
        if word
    ]


    if not words:

        return "B"


    if len(words) == 1:

        return (
            words[0][:2]
            .upper()
        )


    return (
        words[0][0]
        + words[-1][0]
    ).upper()

def draw_shape(
    draw,
    shape,
    cx,
    cy,
    radius,
    color,
    rotation,
    stroke_width,
    outline_only=False,
):

    if shape == "circle":

        box = (
            cx - radius,
            cy - radius,
            cx + radius,
            cy + radius,
        )


        if outline_only:

            draw.ellipse(
                box,
                outline=color,
                width=stroke_width,
            )

        else:

            draw.ellipse(
                box,
                fill=color,
            )


    elif shape == "square":

        points = polygon_points(
            cx,
            cy,
            radius,
            4,
            rotation + 45,
        )


        if outline_only:

            draw.line(
                points + [points[0]],
                fill=color,
                width=stroke_width,
                joint="curve",
            )

        else:

            draw.polygon(
                points,
                fill=color,
            )


    elif shape == "diamond":

        points = polygon_points(
            cx,
            cy,
            radius,
            4,
            rotation,
        )


        if outline_only:

            draw.line(
                points + [points[0]],
                fill=color,
                width=stroke_width,
                joint="curve",
            )

        else:

            draw.polygon(
                points,
                fill=color,
            )


    elif shape == "triangle":

        points = polygon_points(
            cx,
            cy,
            radius,
            3,
            rotation,
        )


        if outline_only:

            draw.line(
                points + [points[0]],
                fill=color,
                width=stroke_width,
                joint="curve",
            )

        else:

            draw.polygon(
                points,
                fill=color,
            )


    elif shape == "hexagon":

        points = polygon_points(
            cx,
            cy,
            radius,
            6,
            rotation,
        )


        if outline_only:

            draw.line(
                points + [points[0]],
                fill=color,
                width=stroke_width,
                joint="curve",
            )

        else:

            draw.polygon(
                points,
                fill=color,
            )


    elif shape == "rings":

        for offset in [
            0,
            45,
            90,
        ]:

            r = max(
                25,
                radius - offset,
            )


            draw.ellipse(
                (
                    cx - r,
                    cy - r,
                    cx + r,
                    cy + r,
                ),
                outline=color,
                width=max(
                    8,
                    stroke_width,
                ),
            )


    elif shape == "arcs":

        draw.arc(
            (
                cx - radius,
                cy - radius,
                cx + radius,
                cy + radius,
            ),
            start=20 + rotation,
            end=170 + rotation,
            fill=color,
            width=stroke_width,
        )


        inner = int(
            radius * 0.65
        )


        draw.arc(
            (
                cx - inner,
                cy - inner,
                cx + inner,
                cy + inner,
            ),
            start=200 + rotation,
            end=350 + rotation,
            fill=color,
            width=stroke_width,
        )


    elif shape == "lines":

        count = 5


        for index in range(count):

            offset = (
                index - 2
            ) * 45


            draw.line(
                (
                    cx - radius,
                    cy + offset,
                    cx + radius,
                    cy - offset,
                ),
                fill=color,
                width=stroke_width,
            )


    elif shape == "waves":

        points = []


        for x in range(
            int(cx - radius),
            int(cx + radius) + 1,
            10,
        ):

            relative = (
                x - cx
            )


            y = (
                cy
                + math.sin(
                    relative / 45
                )
                * radius
                * 0.35
            )


            points.append(
                (
                    x,
                    y,
                )
            )


        draw.line(
            points,
            fill=color,
            width=stroke_width,
        )


        second = [
            (
                x,
                y + 65,
            )
            for x, y
            in points
        ]


        draw.line(
            second,
            fill=color,
            width=max(
                8,
                stroke_width - 5,
            ),
        )


    elif shape == "interlocking":

        r = (
            radius * 0.7
        )


        draw.ellipse(
            (
                cx - r - 55,
                cy - r,
                cx + r - 55,
                cy + r,
            ),
            outline=color,
            width=stroke_width,
        )


        draw.ellipse(
            (
                cx - r + 55,
                cy - r,
                cx + r + 55,
                cy + r,
            ),
            outline=color,
            width=stroke_width,
        )


    else:

        draw.ellipse(
            (
                cx - radius,
                cy - radius,
                cx + radius,
                cy + radius,
            ),
            fill=color,
        )

# =========================================================
# ROUTER
# =========================================================

def generate_image(
    provider=None,
    **kwargs,
):
    """
    Route image generation to the requested provider.

    Supported providers:
        auto
        replicate
        flux
        local
        student
    """

    selected = (
        provider
        or DEFAULT_PROVIDER
    ).lower()


    # -----------------------------------------------------
    # LOCAL
    # -----------------------------------------------------

    if selected == "local":

        return generate_local_image(
            **kwargs
        )


    # -----------------------------------------------------
    # FLUX / POLLINATIONS
    # -----------------------------------------------------

    if selected == "flux":

        return generate_flux_image(
            **kwargs
        )


    # -----------------------------------------------------
    # REPLICATE
    # -----------------------------------------------------

    if selected == "replicate":

        return generate_replicate_provider_image(
            **kwargs
        )


    # -----------------------------------------------------
    # STUDENT
    # -----------------------------------------------------

    if selected == "student":

        return generate_student_image(
            **kwargs
        )


    # -----------------------------------------------------
    # AUTOMATIC FALLBACK
    # -----------------------------------------------------

    if selected == "auto":

        providers = [
            (
                "replicate",
                generate_replicate_provider_image,
            ),
            (
                "flux",
                generate_flux_image,
            ),
        ]

        errors = []


        for name, generator in providers:

            try:

                print(
                    f"Trying image provider: {name}"
                )


                image_bytes = generator(
                    **kwargs
                )


                if not isinstance(
                    image_bytes,
                    (bytes, bytearray),
                ):

                    raise RuntimeError(
                        f"{name} provider returned "
                        "non-binary image data."
                    )


                if len(image_bytes) < 1000:

                    raise RuntimeError(
                        f"{name} provider returned "
                        "invalid image data."
                    )


                print(
                    f"Image provider succeeded: {name}"
                )


                return bytes(
                    image_bytes
                )


            except Exception as error:

                errors.append(
                    f"{name}: {error}"
                )

                print(
                    f"{name} unavailable:",
                    repr(error),
                )


        # ---------------------------------------------
        # FINAL LOCAL FALLBACK
        # ---------------------------------------------

        try:

            print(
                "Trying image provider: local"
            )


            return generate_local_image(
                **kwargs
            )


        except Exception as error:

            errors.append(
                f"local: {error}"
            )


        raise RuntimeError(
            "All image providers failed. "
            + " | ".join(errors)
        )


    # -----------------------------------------------------
    # UNKNOWN PROVIDER
    # -----------------------------------------------------

    raise ValueError(
        f"Unknown image provider: {selected}"
    )

def sanitize_specification(
    specification: dict,
    direction: dict,
    direction_index: int,
) -> dict:

    # -----------------------------------------------------
    # LOGO TYPE
    # -----------------------------------------------------

    preferred_types = direction[
        "preferred_types"
    ]


    logo_type = specification.get(
        "logo_type",
        preferred_types[0],
    )


    if logo_type not in LOGO_TYPES:

        logo_type = preferred_types[0]


    # Force each concept to remain within
    # its intended creative family.
    if logo_type not in preferred_types:

        logo_type = preferred_types[0]


    # -----------------------------------------------------
    # SHAPE
    # -----------------------------------------------------

    preferred_shapes = direction[
        "preferred_shapes"
    ]


    shape = specification.get(
        "shape",
        preferred_shapes[0],
    )


    # Prevent the model from collapsing all
    # three concepts into the same geometry.
    if shape not in preferred_shapes:

        shape = preferred_shapes[0]


    # -----------------------------------------------------
    # LAYOUT
    # -----------------------------------------------------

    layout = specification.get(
        "layout",
        "symbol-top",
    )


    if layout not in LAYOUTS:

        layout = "symbol-top"


    # Wordmarks should use wordmark layout.
    if logo_type == "wordmark":

        layout = "wordmark"


    # -----------------------------------------------------
    # FONT STYLE
    # -----------------------------------------------------

    font_style = specification.get(
        "font_style",
        "modern",
    )


    if font_style not in FONT_STYLES:

        font_style = "modern"

        palette[
            "accent"
        ],
    )


    # -----------------------------------------------------

    return {

# =========================================================
# HEX VALIDATION
# =========================================================

def clean_hex(
    value,
    fallback,
):

    if not isinstance(
        value,
        str,
    ):

        return fallback


    value = value.strip()


    if len(value) != 7:

        return fallback


    if not value.startswith("#"):

        return fallback


    try:

        int(
            value[1:],
            16,
        )


        return value.upper()


    except ValueError:

        return fallback


# =========================================================
# INTEGER VALIDATION
# =========================================================

def clamp_integer(
    value,
    minimum,
    maximum,
    fallback,
):

    try:

        value = int(
            value
        )


        return max(
            minimum,
            min(
                maximum,
                value,
            ),
        )


    except Exception:

        return fallback


# =========================================================
# TEXT VALIDATION
# =========================================================

def clean_text(
    value,
    fallback,
):

    if not isinstance(
        value,
        str,
    ):

        return fallback


    value = value.strip()


    if not value:

        return fallback


    # Prevent extremely verbose local-model outputs
    # from propagating through the renderer.
    return value[:500]

def polygon_points(
    cx,
    cy,
    radius,
    sides,
    rotation=0,
):

    points = []

    for index in range(sides):

        angle = math.radians(
            rotation
            - 90
            + index
            * (360 / sides)
        )

        points.append(
            (
                cx
                + radius
                * math.cos(angle),

                cy
                + radius
                * math.sin(angle),
            )
        )

    return points


def svg_points(
    points,
):

    return " ".join(
        f"{round(x, 2)},{round(y, 2)}"
        for x, y in points
    )


def initials(
    business_name,
):

    words = [
        word
        for word
        in business_name
        .strip()
        .split()
        if word
    ]

    if not words:

        return "B"

    if len(words) == 1:

        return (
            words[0][:2]
            .upper()
        )

    return (
        words[0][0]
        + words[-1][0]
    ).upper()

    if len(name) <= 10:
        size = 86

    elif len(name) <= 16:
        size = 70

        size = 56

    else:
        size = 44

    name_font = get_font(
        size,

        ],
    )

    draw_centered_text(
        draw,
        name,
        600,
        720,
        name_font,
        specification[
            "primary_color"
        ],
    )

    draw.rounded_rectangle(
        (
            530,
            825,
            670,
            833,
        ),
        radius=4,
        fill=specification[
            "secondary_color"
        ],
    )

    industry_text = (
        industry
        .replace(
            "_",
            " "
        )
        .replace(
            "-",
            " "
        )
        .upper()
    )

    industry_font = get_font(
        27,
        "minimal",
    )

    draw_centered_text(
        draw,
        industry_text,
        600,
        860,
        industry_font,
        specification[
            "secondary_color"
        ],
    )

    output = BytesIO()

    image.save(
        output,
        format="PNG",
        optimize=True,
    )

    output.seek(0)

    return output.getvalue()

# =========================================================

def generate_replicate_image(
    specification: dict,
    industry: str,
    style: str,
    purpose: str = "campaign",
) -> bytes:

    token = os.getenv(
        "REPLICATE_API_TOKEN"
    )


    if not token:

        raise RuntimeError(
            "REPLICATE_API_TOKEN is not configured."
        )


    model = os.getenv(
        "REPLICATE_IMAGE_MODEL"
    )


    if not model:

        raise RuntimeError(
            "REPLICATE_IMAGE_MODEL is not configured."
        )


    prompt = _build_prompt(
        specification=specification,
        industry=industry,
        style=style,
        purpose=purpose,
    )


    headers = {
        "Authorization":
        f"Bearer {token}",

        "Content-Type":
        "application/json",

        "Prefer":
        "wait=60",
    }


    response = requests.post(
        (
            f"{REPLICATE_API_URL}/"
            f"models/{model}/predictions"
        ),

        headers=headers,

        json={
            "input": {
                "prompt":
                prompt,
            }
        },

        timeout=90,
    )


    response.raise_for_status()

    prediction = (
        response.json()
    )


    # -----------------------------------------------------
    # POLL UNTIL COMPLETE
    # -----------------------------------------------------

    while prediction.get(
        "status"
    ) not in {
        "succeeded",
        "failed",
        "canceled",
    }:

        poll_url = (
            prediction.get(
                "urls",
                {},
            )
            .get(
                "get"
            )
        )


        if not poll_url:

            raise RuntimeError(
                "Replicate did not provide "
                "a prediction polling URL."
            )


        time.sleep(
            2
        )


        response = requests.get(
            poll_url,

            headers={
                "Authorization":
                f"Bearer {token}",
            },

            timeout=30,
        )


        response.raise_for_status()

# =========================================================
# HEX -> RGB
# =========================================================

def hex_to_rgb(
    value: str,
    fallback=(15, 23, 42),
):

    try:

        value = (
            value
            .strip()
            .lstrip("#")
        )


        if len(value) != 6:

            return fallback


        return tuple(
            int(
                value[index:index + 2],
                16,
            )

            for index
            in (
                0,
                2,
                4,
            )
        )


    except Exception:

        return fallback


# =========================================================
# CONTRAST COLOR
# =========================================================

def text_color_for(
    rgb,
):

    red, green, blue = rgb


    luminance = (
        0.299 * red
        + 0.587 * green
        + 0.114 * blue
    )


    if luminance > 165:

        return (
            15,
            23,
            42,
        )


    return (
        255,
        255,
        255,
    )

def text_width(
    draw,
    text,
    font,
):

    bbox = draw.textbbox(
        (0, 0),
        text,
        font=font,
    )


    return (
        bbox[2]
        - bbox[0]
    )


def draw_centered_text(
    draw,
    text,
    y,
    font,
    fill,
):

    width = text_width(
        draw,
        text,
        font,
    )


    x = (
        WIDTH - width
    ) / 2


    draw.text(
        (
            x,
            y,
        ),
        text,
        font=font,
        fill=fill,
    )


def wrap_text(
    draw,
    text,
    font,
    max_width,
):

    words = (
        str(text or "")
        .split()
    )


    lines = []

    current_line = ""


    for word in words:

        test_line = (
            f"{current_line} {word}"
            .strip()
        )


        width = text_width(
            draw,
            test_line,
            font,
        )


        if width <= max_width:

            current_line = (
                test_line
            )


        else:

            if current_line:

                lines.append(
                    current_line
                )


            current_line = word


    if current_line:

        lines.append(
            current_line
        )


    return lines


def draw_wrapped_center(
    draw,
    text,
    y,
    font,
    fill,
    max_width=850,
    line_height=65,
):

    lines = wrap_text(
        draw,
        text,
        font,
        max_width,
    )


    for line in lines:

        draw_centered_text(
            draw,
            line,
            y,
            font,
            fill,
        )


        y += line_height

def hex_to_rgb(
    value,
    fallback=(15, 23, 42),
):

    try:

        value = (
            value
            .strip()
            .lstrip("#")
        )


        if len(value) != 6:

            return fallback


        return tuple(
            int(
                value[index:index + 2],
                16,
            )

            for index in (
                0,
                2,
                4,
            )
        )


    except Exception:

        return fallback


def mix(
    color_a,
    color_b,
    amount,
):

    return tuple(
        int(
            color_a[index]
            * (1 - amount)
            + color_b[index]
            * amount
        )

        for index in range(3)
    )


def contrast_color(
    color,
):

    luminance = (
        0.299 * color[0]
        + 0.587 * color[1]
        + 0.114 * color[2]
    )


    if luminance > 165:

        return (
            15,
            23,
            42,
        )


    return (
        255,
        255,
        255,
    )

def text_width(
    draw,
    text,
    font,
):

    box = draw.textbbox(
        (0, 0),
        text,
        font=font,
    )


    return (
        box[2]
        - box[0]
    )


def draw_centered(
    draw,
    text,
    y,
    font,
    fill,

    width = text_width(
        draw,
        text,
        font,
    )


    draw.text(
        (
            (WIDTH - width) / 2,

        font=font,

def wrap_text(
    draw,
    text,
    font,
    max_width,
):

    words = (
        str(
            text or ""
        )
        .split()
    )


    lines = []

    current = ""


    for word in words:

        candidate = (
            f"{current} {word}"
            .strip()
        )


        if (
            text_width(
                draw,
                candidate,
                font,
            )
            <= max_width
        ):

            current = candidate


        else:

            if current:

                lines.append(
                    current
                )


            current = word


    if current:

        lines.append(
            current
        )


    return lines


def draw_wrapped(
    draw,

    y,
    font,
    fill,
    max_width=780,
    line_height=65,
):

    lines = wrap_text(
        draw,
        text,
        font,
        max_width,
    )


    for line in lines:

        draw_centered(
            draw,
            line,
            y,
            font,
            fill,
        )


        y += line_height

def animated_image_clip(
    pil_image,
    duration,
    zoom_start=1.0,
    zoom_end=1.04,
):

    base = (
        pil_image
        .convert(
            "RGB"
        )
    )


    def frame_function(t):

        progress = (
            t
            / duration
            if duration
            else 0
        )


        scale = (
            zoom_start
            + (
                zoom_end
                - zoom_start
            )
            * progress
        )


        new_width = int(
            WIDTH * scale
        )


        new_height = int(
            HEIGHT * scale
        )


        frame = base.resize(
            (
                new_width,
                new_height,
            ),
            Image.Resampling.LANCZOS,
        )


        left = (
            new_width
            - WIDTH
        ) // 2


        top = (
            new_height
            - HEIGHT
        ) // 2


        frame = frame.crop(
            (
                left,
                top,
                left + WIDTH,
                top + HEIGHT,
            )
        )


        return np.array(
            frame
        )


    return (
        VideoClip(
            frame_function,
            duration=duration,
        )
    )

                pass

def generate_video(
    provider: str | None = None,
    **kwargs,
) -> Path:

    selected = (
        provider
        or DEFAULT_PROVIDER
    ).lower()


    # -----------------------------------------------------
    # EXPLICIT LOCAL
    # -----------------------------------------------------

    if selected == "local":

        return generate_local_video(
            **kwargs
        )


    # -----------------------------------------------------
    # EXPLICIT WAN
    # -----------------------------------------------------

    if selected == "wan":

        return generate_wan_video(
            **kwargs
        )


    # -----------------------------------------------------
    # EXPLICIT LTX
    # -----------------------------------------------------

    if selected == "ltx":

        return generate_ltx_video(
            **kwargs
        )


    # -----------------------------------------------------
    # AUTO MODE
    # -----------------------------------------------------

    if selected == "auto":

        providers = [
            (
                "wan",
                generate_wan_video,
            ),

            (
                "ltx",
                generate_ltx_video,
            ),
        ]


        for name, generator in providers:

            try:

                print(
                    f"Trying video provider: {name}"
                )


                return generator(
                    **kwargs
                )


            except Exception as error:

                print(
                    f"{name} unavailable:",
                    repr(error),
                )


        print(
            "AI video providers unavailable. "
            "Falling back to local MoviePy V2."
        )


        return generate_local_video(
            **kwargs
        )


    # -----------------------------------------------------
    # INVALID PROVIDER
    # -----------------------------------------------------

    raise ValueError(
        f"Unknown video provider: {selected}"
    )
